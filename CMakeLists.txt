cmake_minimum_required(VERSION 3.16)
project(cherryusb_ipheth VERSION 0.1.0 LANGUAGES C)
option(IPHETH_SANITIZE "Enable address/undefined sanitizers" OFF)
option(IPHETH_TESTS "Build host tests" ON)
add_library(ipheth STATIC src/ipheth.c)
target_include_directories(ipheth PUBLIC include)
target_compile_features(ipheth PUBLIC c_std_99)
target_compile_options(ipheth PRIVATE -Wall -Wextra -Werror)
if(IPHETH_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(ipheth PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(IPHETH_TESTS)
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core PRIVATE ipheth)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME protocol COMMAND test_core)
endif()
