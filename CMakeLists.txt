cmake_minimum_required(VERSION 3.16)
project(cherryusb_ipheth VERSION 0.1.0 LANGUAGES C)
option(IPHETH_SANITIZE "Enable address/undefined sanitizers" OFF)
option(IPHETH_TESTS "Build host tests" ON)
add_library(ipheth STATIC src/ipheth.c)
target_include_directories(ipheth PUBLIC include)
target_compile_features(ipheth PUBLIC c_std_99)
target_compile_options(ipheth PRIVATE -Wall -Wextra -Werror)
if(IPHETH_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  target_compile_options(ipheth PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
if(IPHETH_TESTS)
  enable_testing()
  add_executable(test_core tests/test_core.c)
  target_link_libraries(test_core PRIVATE ipheth)
  target_compile_options(test_core PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  add_test(NAME protocol COMMAND test_core)
endif()

set(CHERRYUSB_ROOT "" CACHE PATH "External CherryUSB checkout for adapter tests")
if(IPHETH_TESTS AND CHERRYUSB_ROOT)
  find_package(Threads REQUIRED)
  add_executable(test_class tests/test_class.c port/cherryusb/usbh_ipheth.c)
  target_include_directories(test_class PRIVATE tests/config
    "${CHERRYUSB_ROOT}/core" "${CHERRYUSB_ROOT}/common" "${CHERRYUSB_ROOT}/class/hub")
  target_compile_definitions(test_class PRIVATE IPHETH_EXTERNAL_NET_ADAPTER)
  target_compile_features(test_class PRIVATE c_std_11)
  target_compile_options(test_class PRIVATE -Wall -Wextra -Werror -UNDEBUG)
  target_link_libraries(test_class PRIVATE ipheth Threads::Threads)
  add_test(NAME usb_lifecycle COMMAND test_class)
  # Keep the real upstream EP0 implementation; discard unrelated host functions.
  add_executable(test_control tests/test_control.c "${CHERRYUSB_ROOT}/core/usbh_core.c")
  target_include_directories(test_control PRIVATE tests/config
    "${CHERRYUSB_ROOT}/core" "${CHERRYUSB_ROOT}/common" "${CHERRYUSB_ROOT}/class/hub")
  target_compile_options(test_control PRIVATE -ffunction-sections -fdata-sections -UNDEBUG)
  target_link_options(test_control PRIVATE -Wl,--gc-sections)
  add_test(NAME upstream_control COMMAND test_control)
endif()
